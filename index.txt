        function searchUser() {
            let friendName = document.getElementById('friendInput').value.trim();
            let resultArea = document.getElementById('searchResultArea');
            resultArea.innerHTML = '';
            document.getElementById('suggestionsBox').style.display = 'none';

            if(!friendName) return;
            if(friendName.toLowerCase() === currentUserNick.toLowerCase()) { 
                resultArea.innerHTML = '<p style="color:var(--danger); text-align:center; font-size:14px; font-weight:bold;">Non puoi cercare te stesso!</p>'; 
                return; 
            }
            
            let exactName = null;
            for (let name in globalLeaderboardScores) {
                if(name.toLowerCase() === friendName.toLowerCase()) { exactName = name; break; }
            }
            
            if(exactName) {
                let actionsHtml = '';
                if (myFriendsList.includes(exactName)) { 
                    actionsHtml = `<span class="badge-status" style="color:var(--social-blue);">🤝 Già Amici</span>`; 
                } 
                else if (mySentRequests.includes(exactName)) { 
                    actionsHtml = `<span class="badge-status" style="color:var(--text-muted);">⏳ In attesa di risposta</span>`; 
                } 
                else if (myReceivedRequests.includes(exactName)) { 
                    actionsHtml = `<button class="btn-action btn-accept" onclick="navigateTo('page-friends')">Rispondi alla sua richiesta</button>`; 
                } 
                else { 
                    actionsHtml = `<button class="btn-action btn-add" onclick="sendFriendRequestFromSearch('${exactName}')">➕ Invia Richiesta</button>`; 
                }

                resultArea.innerHTML = `
                    <div class="search-result-card">
                        <div style="font-weight: bold; font-size:18px; color:var(--dark);">👤 ${exactName}</div>
                        <div>${actionsHtml}</div>
                    </div>
                `;
            } else {
                resultArea.innerHTML = '<p style="color:var(--danger); text-align:center; font-size:14px; font-weight:bold;">Utente non trovato nel database.</p>';
            }
        }

        function acceptFriendRequest(senderNick) {
            db.ref('users_friends/' + currentUserNick + '/' + senderNick).set(true);
            db.ref('users_friends/' + senderNick + '/' + currentUserNick).set(true);
            db.ref('friend_requests/' + currentUserNick + '/' + senderNick).remove();

            db.ref('friend_requests_sent/' + senderNick + '/' + currentUserNick).remove();

        }

        function cancelSentRequest(targetNick) {

                db.ref('users_friends/' + friendNick + '/' + currentUserNick).remove();
            }
        }

        function restoreRejectedRequest(senderNick) {

        function renderFriendsManager() {
            const list = document.getElementById('friendsManagerList'); list.innerHTML = '';

                myReceivedRequests.forEach(sender => {
                    let item = document.createElement('div'); item.className = 'list-item';
                    item.innerHTML = `<div class="lb-user">📩 ${sender}</div><div style="display:flex; gap:10px;"><button class="btn-action btn-accept" onclick="acceptFriendRequest('${sender}')">✔️ Accetta</button><button class="btn-action btn-reject" onclick="rejectFriendRequest('${sender}')">❌ Rifiuta</button></div>`;
                    list.appendChild(item);
                });
            } 
            else if (currentFriendsTab === 'sent') {
                if (mySentRequests.length === 0) { list.innerHTML = '<p style="text-align:center;">Nessuna richiesta inviata.</p>'; return; }
                mySentRequests.forEach(target => {
                    let item = document.createElement('div'); item.className = 'list-item';

                    list.appendChild(item);

            else if (currentFriendsTab === 'accepted') {
                if (myFriendsList.length === 0) { list.innerHTML = '<p style="text-align:center;">Non hai ancora aggiunto amici.</p>'; return; }

                    let item = document.createElement('div'); item.className = 'list-item';

                });

            else if (currentFriendsTab === 'rejected') {
                if (myRejectedRequests.length === 0) { list.innerHTML = '<p style="text-align:center;">Nessun utente rifiutato.</p>'; return; }
                myRejectedRequests.forEach(sender => {
                    let item = document.createElement('div'); item.className = 'list-item';
                    item.innerHTML = `<div class="lb-user">❌ ${sender}</div><button class="btn-action btn-accept" onclick="restoreRejectedRequest('${sender}')">Sblocca e Accetta</button>`;
                    list.appendChild(item);
                });
            }
        }

        function handlePhotoUpload(event, brand, variant) {
            event.stopPropagation(); const file = event.target.files[0]; if (!file) return;
            if (!aiModel) { alert("L'AI si sta configurando, riprova tra poco!"); return; }

            const reader = new FileReader();
            reader.onload = function(e) {
                const img = new Image();
                img.onload = function() {
                    const canvas = document.createElement('canvas'); const MAX_SIZE = 400;
                    let width = img.width; let height = img.height;
                    if (width > height) { if (width > MAX_SIZE) { height *= MAX_SIZE / width; width = MAX_SIZE; } } else { if (height > MAX_SIZE) { width *= MAX_SIZE / height; height = MAX_SIZE; } }
                    canvas.width = width; canvas.height = height;
                    const ctx = canvas.getContext('2d'); ctx.drawImage(img, 0, 0, width, height);

                    const imgData = ctx.getImageData(0, 0, width, height); const data = imgData.data; let totalBrightness = 0;
                    for (let i = 0; i < data.length; i += 4) { totalBrightness += (data[i] + data[i+1] + data[i+2]) / 3; }
                    if ((totalBrightness / (data.length / 4)) < 20) { alert("🚨 FOTO RESPINTA: Immagine troppo scura!"); return; }

                    document.getElementById('progressText').innerText = "🤖 Analisi Forma (AI) in corso...";

                    aiModel.detect(canvas).then(predictions => {
                        let isDrink = false; predictions.forEach(p => { if(p.class === 'bottle' || p.class === 'cup' || p.class === 'wine glass') isDrink = true; });

                        if (!isDrink) { alert("🚨 AI ANTI-CHEAT: Foto respinta! Non vedo bottiglie o bicchieri veri."); updateStats(); return; }

                        // Step 2: Lettura OCR Etichetta tramite Tesseract.js
                        document.getElementById('progressText').innerText = "🔍 Lettura etichetta in corso...";
                        
                        Tesseract.recognize(
                            canvas,
                            'ita+eng',
                            { logger: m => console.log(m) }
                        ).then(({ data: { text } }) => {
                            const extractedText = text.toLowerCase();
                            // Prendiamo la prima parola del brand per fare un match più flessibile (es. "Mastri" per "Mastri Birrai Umbri")
                            const brandWord = brand.toLowerCase().split(' ')[0];
                            
                            // Controlliamo se Tesseract ha letto qualcosa simile al brand nella foto
                            if (extractedText.includes(brandWord) || extractedText.replace(/\s/g, '').includes(brandWord)) {
                                
                                // Successo! L'etichetta è verificata
                                const compressedDataUrl = canvas.toDataURL('image/jpeg', 0.6);
                                const uniqueId = brand + '-' + variant;
                                
                                db.ref('pokedex_profiles/' + currentUserNick + '/' + uniqueId).set(compressedDataUrl).then(() => {
                                    const newPostRef = db.ref('social_timeline').push();
                                    newPostRef.set({ user: currentUserNick, brand: brand, variant: variant, photo: compressedDataUrl, time: new Date().getTime() }).then(() => {
                                        let currentScore = Object.keys(myPokedex).length;
                                        db.ref('leaderboard_scores/' + currentUserNick).set(currentScore);
                                        alert("✅ Birra catturata, verificata e letta con successo!");
                                    });
                                });
                            } else {
                                alert(`🚨 FOTO RESPINTA: L'AI non riesce a leggere la scritta "${brand}" sull'etichetta. Metti a fuoco meglio!`);
                                updateStats(); // resetta il testo
                            }
                        });
                    });
                }
                img.src = e.target.result;
            }
            reader.readAsDataURL(file);
        }

        function deleteBeerPhoto(brand, variant) {
            const uniqueId = brand + '-' + variant;
            if(!confirm(`Vuoi davvero eliminare la foto di ${brand} - ${variant}?`)) return;

            db.ref('pokedex_profiles/' + currentUserNick + '/' + uniqueId).remove().then(() => {
                let newScore = Object.keys(myPokedex).length;
                db.ref('leaderboard_scores/' + currentUserNick).set(newScore);

                db.ref('social_timeline').once('value').then(snapshot => {
                    snapshot.forEach(child => {
                        let post = child.val();
                        if(post.user === currentUserNick && post.brand === brand && post.variant === variant) { db.ref('social_timeline/' + child.key).remove(); }
                    });
                });
            });
        }

        function generateVariantsHTML(brand, variantsArray) {
            let listHTML = '';
            variantsArray.forEach(variant => {
                let uniqueId = brand + '-' + variant; let hasPhoto = myPokedex[uniqueId] !== undefined;
                listHTML += `
                <div class="variant-item" onclick="event.stopPropagation()">
                    <span style="flex-grow: 1; font-weight: 500;">${variant}</span>
                    ${hasPhoto ? `
                        <div class="unlocked-status"><span>✔ Sbloccato</span><img src="${myPokedex[uniqueId]}" class="thumb-preview"><button class="btn-delete" onclick="deleteBeerPhoto('${brand}', '${variant}')" title="Elimina foto">🗑️</button></div>
                    ` : `
                        <label class="photo-btn" onclick="event.stopPropagation()">📷 Scatta<input type="file" accept="image/*" capture="environment" onclick="event.stopPropagation()" onchange="handlePhotoUpload(event, '${brand}', '${variant}')"></label>
                    `}
                </div>`;
            });
            return listHTML;
        }

        function displayBeers(beerArray) {
            const container = document.getElementById('beerList'); container.innerHTML = '';
            beerArray.forEach((beer, index) => {
                let realIndex = beers.findIndex(b => b.brand === beer.brand);
                const card = document.createElement('div'); card.className = 'card'; card.id = 'card-' + realIndex; card.setAttribute('onclick', 'toggleCard(this)');
                card.innerHTML = `

                    <h2>${beer.brand} <span class="trophy" style="display:none;">🏆</span></h2><div class="desc">${beer.desc}</div>

                `;
                container.appendChild(card);
            });
            checkTrophiesVisually();
        }

        function updateStats() {
            let totalBeers = 0; let totalDrunk = 0;
            beers.forEach((beer) => { totalBeers += beer.variants.length; beer.variants.forEach(variant => { if (myPokedex[beer.brand + '-' + variant]) { totalDrunk++; } }); });
            const percentage = totalBeers === 0 ? 0 : Math.round((totalDrunk / totalBeers) * 100);
            document.getElementById('progressBar').style.width = percentage + '%';
            document.getElementById('progressText').innerText = `Hai collezionato ${totalDrunk} su ${totalBeers} (${percentage}%)`;
            checkTrophiesVisually();
        }

        function checkTrophiesVisually() {
            beers.forEach((beer, index) => {
                let brandTotal = beer.variants.length; let brandDrunk = 0;
                beer.variants.forEach(variant => { if (myPokedex[beer.brand + '-' + variant]) { brandDrunk++; } });
                const card = document.getElementById('card-' + index);
                if (card) { if (brandDrunk === brandTotal && brandTotal > 0) { card.classList.add('completed'); card.querySelector('.trophy').style.display = 'inline-block'; } else { card.classList.remove('completed'); card.querySelector('.trophy').style.display = 'none'; } }
            });
        }

        function renderTrophies() {
            const grid = document.getElementById('trophyGrid'); grid.innerHTML = '';
            beers.forEach(beer => {
                beer.variants.forEach(variant => {
                    let uniqueId = beer.brand + '-' + variant; let photoUrl = myPokedex[uniqueId]; let isUnlocked = photoUrl !== undefined;

                    trophyDiv.innerHTML = `<div class="trophy-brand">${beer.brand}</div><div class="trophy-img-container">${imageContent}</div><div class="trophy-name">${variant}</div>`;
                    grid.appendChild(trophyDiv);
                });
            });
        }

        function renderLeaderboard() {
            const list = document.getElementById('leaderboardList'); list.innerHTML = '';
            
            let players = [];
            for (let name in globalLeaderboardScores) { 
                if(currentLeaderboardTab === 'friends') { if(name !== currentUserNick && !myFriendsList.includes(name)) continue; }
                players.push({ name: name, score: globalLeaderboardScores[name] }); 
            }
            players.sort((a, b) => b.score - a.score);
            
            if(players.length === 0) { list.innerHTML = '<p style="text-align:center;">Nessun punteggio.</p>'; return; }
            
            players.forEach((player, index) => {
                let item = document.createElement('div'); item.className = 'leaderboard-item';
                let medal = (index + 1); if(index === 0) medal = '🥇'; else if(index === 1) medal = '🥈'; else if(index === 2) medal = '🥉';
                
                let actionsHtml = '';
                if (currentLeaderboardTab === 'global' && player.name !== currentUserNick) {
                    if (myFriendsList.includes(player.name)) { actionsHtml = `<span class="badge-status" style="color:var(--social-blue);">🤝 Amici</span>`; } 
                    else if (mySentRequests.includes(player.name)) { actionsHtml = `<span class="badge-status" style="color:var(--text-muted);">⏳ In attesa</span>`; } 
                    else if (myReceivedRequests.includes(player.name)) { actionsHtml = `<button class="btn-action btn-accept" onclick="navigateTo('page-friends')">Rispondi</button>`; } 
                    else { actionsHtml = `<button class="btn-action btn-add" onclick="sendFriendRequestFromSearch('${player.name}')" style="padding:5px 10px; font-size:12px;">➕ Aggiungi</button>`; }
                }
                item.innerHTML = `<div style="display:flex; align-items:center; flex-grow:1;"><div class="rank">${medal}</div><div class="lb-user">${player.name} ${player.name === currentUserNick ? '(Tu)' : ''}</div></div><div style="display:flex; align-items:center; gap:15px;"><div class="lb-score">${player.score} 🏆</div>${actionsHtml}</div>`;
                list.appendChild(item);
            });
        }
